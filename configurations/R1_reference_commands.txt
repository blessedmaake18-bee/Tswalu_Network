! =====================================================================
! R1 (Cisco 1941) - REFERENCE COMMAND SCRIPT
! Tswalu Steel Fabrication - CMPG325 - B. Maake (41432363)
!
! NOTE: this is a command script built from the project design. It is
! NOT an export from the device. Your evidence file
! Router_R1_Config.txt must be a real "show running-config" export
! (see the export steps in the chat). Fill in the <PLACEHOLDERS> if you
! use this script to rebuild the router.
! =====================================================================
enable
configure terminal
hostname R1
no ip domain-lookup
ip domain-name <DOMAIN_NAME>
service password-encryption
enable secret <ENABLE_SECRET>
username netadmin secret <ADMIN_PASSWORD>
ipv6 unicast-routing
!
! ---- Guest isolation ACLs ----
ip access-list extended GUEST-ISOLATION
 deny ip 172.30.12.96 0.0.0.31 172.30.12.64 0.0.0.31
 deny ip 172.30.12.96 0.0.0.31 172.30.12.0 0.0.0.63
 deny ip 172.30.12.96 0.0.0.31 172.30.12.128 0.0.0.15
 deny ip 172.30.12.96 0.0.0.31 172.30.12.144 0.0.0.15
 permit ip any any
exit
ipv6 access-list GUEST-ISOLATION-V6
 deny ipv6 2001:DB8:ACAD:50::/64 2001:DB8:ACAD:10::/64
 deny ipv6 2001:DB8:ACAD:50::/64 2001:DB8:ACAD:20::/64
 deny ipv6 2001:DB8:ACAD:50::/64 2001:DB8:ACAD:30::/64
 deny ipv6 2001:DB8:ACAD:50::/64 2001:DB8:ACAD:40::/64
 permit ipv6 any any
exit
!
! ---- Router-on-a-stick sub-interfaces ----
interface GigabitEthernet0/0
 no shutdown
exit
interface GigabitEthernet0/0.10
 description Management and Administration
 encapsulation dot1Q 10
 ip address 172.30.12.65 255.255.255.224
 ipv6 address 2001:DB8:ACAD:10::1/64
exit
interface GigabitEthernet0/0.20
 description Production
 encapsulation dot1Q 20
 ip address 172.30.12.1 255.255.255.192
 ipv6 address 2001:DB8:ACAD:20::1/64
exit
interface GigabitEthernet0/0.30
 description Finance and HR
 encapsulation dot1Q 30
 ip address 172.30.12.129 255.255.255.240
 ipv6 address 2001:DB8:ACAD:30::1/64
exit
interface GigabitEthernet0/0.40
 description IT and Services
 encapsulation dot1Q 40
 ip address 172.30.12.145 255.255.255.240
 ipv6 address 2001:DB8:ACAD:40::1/64
exit
interface GigabitEthernet0/0.50
 description Guest Wi-Fi
 encapsulation dot1Q 50
 ip address 172.30.12.97 255.255.255.224
 ipv6 address 2001:DB8:ACAD:50::1/64
 ip access-group GUEST-ISOLATION in
 ipv6 traffic-filter GUEST-ISOLATION-V6 in
exit
!
! ---- SSH (run once, answer the modulus prompt with 2048) ----
crypto key generate rsa
ip ssh version 2
!
! ---- Secured lines ----
line console 0
 login local
 exec-timeout 5 0
exit
line vty 0 15
 login local
 transport input ssh
 exec-timeout 5 0
exit
end
copy running-config startup-config
!
! If your trunk uses a native VLAN other than 1 (for example VLAN 99),
! add this sub-interface so both ends match, and create the VLAN on SW1:
!   interface GigabitEthernet0/0.99
!    encapsulation dot1Q 99 native
